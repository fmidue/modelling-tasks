\newcommand\classbox@[6]{
  \node[rectangle split,rectangle split parts=3,draw,text width=#1,
        anchor=north]
    (#6) at #2 {
      \textbf{#3}
      \nodepart{second}
      \texttt{\small #4}
      \nodepart{third}
      \texttt{\small #5}
    } ;
}
\newcommand\classbox[5][4cm]{
  \classbox@{#1}{#2}{\textbf{#3}}{#4}{#5}{#3}
}
\newcommand\virtualclassbox[5][4cm]{
  \classbox@{#1}{#2}{\textbf{\textit{#3}}}{#4}{#5}{#3}
}
